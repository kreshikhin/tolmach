Imports System.IO

Public Class Backup
    Public Sub CopyFolder(source As String, dest As String)
        If Not Directory.Exists(source) Then
            Throw New DirectoryNotFoundException(source)
        End If
        Directory.CreateDirectory(dest)
        For Each file As String In Directory.GetFiles(source)
            Dim name As String = Path.GetFileName(file)
            File.Copy(file, Path.Combine(dest, name), True)
        Next
    End Sub
End Class
