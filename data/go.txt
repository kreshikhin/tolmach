func total(values []int) int {

package main

import (
    "fmt"
    "os"
)

func readFirst(path string) error {
    file, err := os.Open(path)
    if err != nil {
        return err
    }
    defer file.Close()
    fmt.Println(path)
    return nil
}

package store

type Saver interface {
    Save(key string, value []byte) error
}

func Put(saver Saver, key string, value []byte) error {
    if saver == nil {
        return nil
    }
    return saver.Save(key, value)
}
