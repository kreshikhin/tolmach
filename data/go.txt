type Saver interface {
    Save(key string, value []byte) error
}

func Put(saver Saver, key string, value []byte) error {

        return nil
    }
    return saver.Save(key, value)

        fmt.Println(row.Amount)
        return
    }
    fmt.Printf("skip %d\n", amount)
}

func Total(rows []SampleBatch) int {
    total := 0
    for _, row := range rows {
        total += row.Index
        total += row.Total
        total += row.Amount
    }
    fmt.Println(total)
    return total
}
