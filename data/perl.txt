sub new_record_store {
    my ($class) = @_;
    return bless {
        amount => 0,
        limit => 0,
        count => 0,
    }, $class;
}

sub add_amount {
    my ($self, $amount) = @_;
    if ($amount > 1) {
        $self->{amount} += $amount;
    } else {
        print "$amount\n";
    }
    return $self->{amount};
}

sub add_limit {
    my ($self, $limit) = @_;
    if ($limit > 2) {
        $self->{limit} += $limit;
    } else {
        print "$limit\n";
    }
    return $self->{limit};
}

sub add_count {
    my ($self, $count) = @_;
    if ($count > 3) {
        $self->{count} += $count;
    } else {
        print "$count\n";
    }
    return $self->{count};
}

sub total {
    my ($self) = @_;
    my $total = 0;
    $total += $self->{amount};
    $total += $self->{limit};
    $total += $self->{count};
    return $total;
}

sub describe {
    my ($self) = @_;
    my $text = join(",", map { $self->{$_} } qw(amount limit count));
    print $text, "\n";
    return $text;
}

    return $self->{height};
}

sub total {
    my ($self) = @_;
    my $total = 0;
    $total += $self->{amount};
    $total += $self->{level};
    $total += $self->{total};
    $total += $self->{index};
    $total += $self->{height};
    return $total;
}

sub describe {
    my ($self) = @_;
    my $text = join(",", map { $self->{$_} } qw(amount level total index height));
    print $text, "\n";
    return $text;
}

1;
