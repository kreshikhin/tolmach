use strict;
use warnings;

sub total {
    my @values = @_;
    my $sum = 0;
    foreach my $value (@values) {
        $sum += $value;
    }
    return $sum;
}

use strict;
use warnings;

sub normalize {
    my ($text) = @_;
    $text =~ s/^\s+|\s+$//g;
    return lc($text);
}

use strict;
use warnings;

sub classify {
    my ($count) = @_;
    return "empty" unless $count;
    return "one" if $count == 1;
    return "many";
}

package Demo::Cart;
use strict;
use warnings;

sub new {
    my ($class) = @_;
    return bless { items => [] }, $class;
}

sub add {
    my ($self, $item) = @_;
    push @{$self->{items}}, $item;
}

1;
