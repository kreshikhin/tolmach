factorial :: Integer -> Integer
factorial 0 = 1
factorial n = n * factorial (n - 1)

data User = User { name :: String, age :: Int }
    deriving (Show, Eq)

greet :: User -> String
greet user = "hello " ++ name user

---
keepPositive :: [Int] -> [Int]
keepPositive values = filter (> 0) values

doubleAll :: [Int] -> [Int]
doubleAll values = map (* 2) values

---
sumList :: Num a => [a] -> a
sumList [] = 0
sumList (x:xs) = x + sumList xs

describe :: Int -> String
describe n
    | n == 0 = "zero"
    | n < 0 = "negative"
    | otherwise = "positive"
