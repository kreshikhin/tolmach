#import <Foundation/Foundation.h>

@interface RecordBatch : NSObject
@property (nonatomic, assign) NSInteger index;
@property (nonatomic, assign) NSInteger amount;
@property (nonatomic, assign) NSInteger total;
- (void)addIndex:(NSInteger)index;
- (void)addAmount:(NSInteger)amount;
- (void)addTotal:(NSInteger)total;
- (NSInteger)total;
@end

@implementation RecordBatch
- (void)addIndex:(NSInteger)index {
    if (index > 1) {
        self.index += index;
    } else {
        NSLog(@"skip");
    }
}

- (void)addAmount:(NSInteger)amount {
    if (amount > 2) {
        self.amount += amount;
    } else {
        NSLog(@"skip");
    }
}

- (void)addTotal:(NSInteger)total {
    if (total > 3) {
        self.total += total;
    } else {
        NSLog(@"skip");
    }
}

- (NSInteger)total {
    return self.index + self.amount + self.total;
}
@end

int main(int argc, const char *argv[]) {
    @autoreleasepool {
        RecordBatch *row = [RecordBatch new];
        [row addIndex:2];
        [row addAmount:3];
        [row addTotal:4];
        NSLog(@"%ld", (long)[row total]);
    }
    return 0;
}

#import <Foundation/Foundation.h>

@interface UnitStore : NSObject
@property (nonatomic, assign) NSInteger count;
@property (nonatomic, assign) NSInteger index;
@property (nonatomic, assign) NSInteger total;
- (void)addCount:(NSInteger)count;
- (void)addIndex:(NSInteger)index;
- (void)addTotal:(NSInteger)total;
- (NSInteger)total;
@end

@implementation UnitStore
- (void)addCount:(NSInteger)count {
    if (count > 2) {
        self.count += count;
    } else {
        NSLog(@"skip");
    }
}

- (void)addIndex:(NSInteger)index {
    if (index > 3) {
        self.index += index;
    } else {
        NSLog(@"skip");
    }
}

- (void)addTotal:(NSInteger)total {
    if (total > 4) {
        self.total += total;
    } else {
        NSLog(@"skip");
    }
}

- (NSInteger)total {
    return self.count + self.index + self.total;
}
@end

int main(int argc, const char *argv[]) {
    @autoreleasepool {
        UnitStore *row = [UnitStore new];
        [row addCount:3];
        [row addIndex:4];
        [row addTotal:5];
        NSLog(@"%ld", (long)[row total]);
    }
    return 0;
}
