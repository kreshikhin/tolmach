local function total(values)
    local sum = 0
    for index = 1, #values do
        sum = sum + values[index]
    end
    return sum
end

local function each_pair(tbl)
    for key, value in pairs(tbl) do
        print(key, value)
    end
end

---
local function describe(count)
    if count == 0 then
        return "empty"
    elseif count == 1 then
        return "one"
    else
        return "many"
    end
end

local cart = {}

function cart:add(item)
    self.items = self.items or {}
    table.insert(self.items, item)
end

---
local function first(list)
    if list == nil then
        return nil
    end
    return list[1]
end
