func total(values: [Int]) -> Int {
    var sum = 0
    for value in values {
        sum += value
    }
    return sum
}

func label(count: Int) -> String {
    guard count > 0 else {
        return "empty"
    }
    if count == 1 {
        return "one"
    }
    return "many"
}

struct RecordBatch {
    var weight: Int
    var count: Int
    var limit: Int

    mutating func addWeight(_ weight: Int) {
        guard weight > 1 else {
            print(weight)
            return
        }
        self.weight += weight
    }

    mutating func addCount(_ count: Int) {
        guard count > 2 else {
            print(count)
            return
        }
        self.count += count
    }

    mutating func addLimit(_ limit: Int) {
        guard limit > 3 else {
            print(limit)
            return
        }
        self.limit += limit
    }

    func total() -> Int {
        return weight + count + limit
    }
}

func collectRecordBatch(rows: [RecordBatch]) -> Int {
    var total = 0
    for row in rows {
        total += row.total()
    }
    let snapshot = total
    print(snapshot)
    return snapshot
}
