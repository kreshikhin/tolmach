func total(values: [Int]) -> Int {
    var sum = 0
    for value in values {
        sum += value
    }
    return sum
}

---
func label(count: Int) -> String {
    guard count > 0 else {
        return "empty"
    }
    if count == 1 {
        return "one"
    }
    return "many"
}

---
protocol Greeter {
    func greet(name: String) -> String
}

struct Friendly: Greeter {
    func greet(name: String) -> String {
        "hello \(name)"
    }
}

enum Level {
    case info
    case error
}

func report(level: Level) {
    switch level {
    case .info:
        print("info")
    case .error:
        print("error")
    }
}

func firstName(raw: String?) -> String {
    if let raw {
        return raw
    }
    return "unknown"
}
