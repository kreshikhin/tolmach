program hello
    implicit none
    print *, "hello"
end program hello

---
subroutine add_one(value)
    implicit none
    integer, intent(inout) :: value
    value = value + 1
end subroutine add_one

---
program total
    implicit none
    integer :: i, sum
    sum = 0
    do i = 1, 10
        sum = sum + i
    end do
    print *, sum
end program total

---
module geometry
    implicit none
contains
    function circle_area(radius) result(area)
        real, intent(in) :: radius
        real :: area
        area = 3.14159 * radius * radius
    end function circle_area
end module geometry

---
program allocate_demo
    implicit none
    integer, allocatable :: values(:)
    integer :: i
    allocate(values(4))
    do i = 1, size(values)
        values(i) = i * i
    end do
    print *, values
    deallocate(values)
end program allocate_demo

---
program branches
    implicit none
    integer :: temperature
    temperature = 30
    if (temperature > 25) then
        print *, "warm"
    else
        print *, "cool"
    end if
end program branches

---
subroutine scale(values, factor)
    implicit none
    real, intent(inout) :: values(:)
    real, intent(in) :: factor
    integer :: i
    do i = 1, size(values)
        values(i) = values(i) * factor
    end do
end subroutine scale
