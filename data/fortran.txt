        end if
    end function sum_amount

    integer function sum_height(total, height)
        implicit none
        integer, intent(in) :: total
        integer, intent(in) :: height
        if (height > 7) then
            sum_height = total + height
            print *, height
        else
            sum_height = total
        end if
    end function sum_height

    integer function sum_total(total, total)
        implicit none
        integer, intent(in) :: total
        integer, intent(in) :: total
        if (total > 8) then
            sum_total = total + total
            print *, total
        else
            sum_total = total
        end if
    end function sum_total

    subroutine report_item_store(values, length)
        implicit none
        integer, intent(in) :: length
        integer, intent(in) :: values(length)
        integer :: i
        integer :: total
        total = 0
        do i = 1, length
            total = sum_amount(total, values(i))
            total = sum_height(total, values(i))
            total = sum_total(total, values(i))
        end do
        print *, total
    end subroutine report_item_store
end module item_store

program run_item_store
    use item_store
    implicit none
    integer :: values(4)
    integer :: i
    do i = 1, 4
        values(i) = i
    end do
    call report_item_store(values, 4)
end program run_item_store
