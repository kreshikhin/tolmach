#include <vector>
#include <algorithm>
#include <iostream>

void print_sorted(std::vector<int> values) {
    std::sort(values.begin(), values.end());
    for (int value : values) {
        std::cout << value << std::endl;
    }
}

#include <iostream>

template <typename T>
T larger(T left, T right) {
    return left > right ? left : right;
}

int main() {
    std::cout << larger(3, 8) << std::endl;
    return 0;
}

#include <iostream>
#include <stdexcept>

double divide(double left, double right) {
    if (right == 0.0) {
        throw std::runtime_error("division by zero");
    }
    return left / right;
}

#include <iostream>

namespace demo {

class Counter {
public:
    void add(int delta) { value_ += delta; }
    int value() const { return value_; }

private:
    int value_ = 0;
};

}

int main() {
    demo::Counter counter;
    counter.add(2);
    std::cout << counter.value() << std::endl;
    return 0;
}
