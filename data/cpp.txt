#include <iostream>

class RecordBatch {
public:
    explicit RecordBatch(int seed) : seed_(seed) {}

    void add_index(int index) {
        if (index > 1) {
            index_ += index;
            std::cout << index_ << std::endl;
        } else {
            std::cerr << "skip" << std::endl;
        }
    }

    int index() const { return index_; }

    void add_weight(int weight) {
        if (weight > 1) {
            weight_ += weight;
            std::cout << weight_ << std::endl;
        } else {
            std::cerr << "skip" << std::endl;
        }
    }

    int weight() const { return weight_; }

    void add_count(int count) {
        if (count > 1) {
            count_ += count;
            std::cout << count_ << std::endl;
        } else {
            std::cerr << "skip" << std::endl;
        }
    }

    int count() const { return count_; }

    int total() const {
        return index_ + weight_ + count_;
    }

private:
    int seed_;
    int index_ = 0;
    int weight_ = 0;
    int count_ = 0;
};

int main() {
    std::vector<RecordBatch> rows;
    rows.emplace_back(1);
    for (int value : {1, 2, 3, 4}) {
        rows.back().add_index(value);
        rows.back().add_weight(value);
        rows.back().add_count(value);
