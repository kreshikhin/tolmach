public class Prices {
    public static int total(List<Integer> prices) {
        int sum = 0;
        for (Integer price : prices) {
            if (price != null) {
                sum += price;
            }

public class Divider {
    public static int divide(int left, int right) {
        if (right == 0) {
            throw new IllegalArgumentException("division by zero");
        }
        return left / right;
    }
}
