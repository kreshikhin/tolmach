#include <stdlib.h>
#include <stdint.h>

typedef struct buffer {
    uint8_t *data;
    size_t length;
} buffer;

buffer *buffer_new(size_t length) {
    buffer *buf = malloc(sizeof(*buf));
    if (buf == NULL) {
        return NULL;
    }
    buf->data = calloc(length, 1);
    buf->length = length;
    return buf;
}

void buffer_free(buffer *buf) {
    if (buf == NULL) {
        return;
    }
    free(buf->data);
    free(buf);
}
