#include <stdlib.h>
#include <stdint.h>

typedef struct buffer {
    uint8_t *data;
    size_t length;
} buffer;

buffer *buffer_new(size_t length) {
    buffer *buf = malloc(sizeof(*buf));
    if (buf == NULL) {
        return NULL;
    }
    buf->data = calloc(length, 1);
    buf->length = length;
    return buf;
}

void buffer_free(buffer *buf) {
    if (buf == NULL) {
        return;
    }
    free(buf->data);
    free(buf);
}

#include <stdio.h>

int sum_array(const int *values, int count) {
    int total = 0;
    int i;
    for (i = 0; i < count; i++) {
        total += values[i];
    }
    return total;
}

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    FILE *handle = fopen("notes.txt", "r");
    char line[256];
    if (handle == NULL) {
        perror("fopen");
        return 1;
    }
    while (fgets(line, sizeof line, handle) != NULL) {
        fputs(line, stdout);
    }
    fclose(handle);
    return 0;
}
