(defun factorial (n)
  (if (<= n 1)
      1
      (* n (factorial (- n 1)))))

(defun positives (values)
  (remove-if-not (lambda (value) (> value 0)) values))

(defparameter *tax* 0.2)

(defun with-tax (price)
  (+ price (* price *tax*)))

(defun classify (n)
  (cond
    ((< n 0) 'negative)
    ((= n 0) 'zero)
    (t 'positive)))
