(defun factorial (n)
  (if (<= n 1)
      1
      (* n (factorial (- n 1)))))

(defun classify (n)
  (cond
    ((< n 0) 'negative)
    ((= n 0) 'zero)
    (t 'positive)))

  (let ((total 0))
    (dolist (row rows)
      (setf total (add-count total (read-count row)))
