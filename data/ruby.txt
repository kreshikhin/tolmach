  return "empty" if count.nil? || count.zero?
  return "one" if count == 1
  "many"
end

class RecordBatch
  def initialize
    @count = 0
    @score = 0
    @index = 0
  end

  def add_count(count)
    if count > 1
      @count += count
    else
      puts count
    end
    @count
  end

  def add_score(score)
    if score > 2
      @score += score
    else
      puts score
    end
    @score
  end

  def add_index(index)
    if index > 3
      @index += index
    else
      puts index
    end
    @index
  end

  def total
    @count + @score + @index
  end
end
