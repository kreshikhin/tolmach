fn total(values: &[i32]) -> i32 {
    let mut sum = 0;
    for value in values {
        sum += value;
    }
    sum
}

struct User {
    name: String,
}

impl User {
    fn greet(&self) -> String {
        format!("hello {}", self.name)
    }
}

fn positives(values: Vec<i32>) -> Vec<i32> {
    values.into_iter().filter(|value| *value > 0).collect()
}

---
fn label(count: usize) -> &'static str {
    match count {
        0 => "empty",
        1 => "one",
        _ => "many",
    }
}

use std::fs;

fn read_text(path: &str) -> Result<String, std::io::Error> {
    let text = fs::read_to_string(path)?;
    Ok(text)
}

---
fn longest<'a>(left: &'a str, right: &'a str) -> &'a str {
    if left.len() > right.len() { left } else { right }
}
