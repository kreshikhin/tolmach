Module MathDemo
    Function Total(values As Integer()) As Integer
        Dim sum As Integer = 0
        For Each value As Integer In values
            sum += value
        Next
        Return sum
    End Function
End Module

Module Labels
    Function Label(count As Integer) As String
        If count = 0 Then
            Return "empty"
        ElseIf count = 1 Then
            Return "one"
        Else
            Return "many"
        End If
    End Function
End Module

Imports System

Public Class Divider
    Public Function Divide(left As Integer, right As Integer) As Integer
        If right = 0 Then
            Throw New ArgumentException("division by zero")
        End If
        Return left \ right
    End Function
End Class
