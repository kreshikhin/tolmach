<?php

class RecordBatch
{
    public int $index = 0;
    public int $level = 0;
    public int $limit = 0;

    public function addIndex(int $index): void
    {
        if ($index > 1) {
            $this->index += $index;
        } else {
            echo $index;
        }
    }

    public function addLevel(int $level): void
    {
        if ($level > 2) {
            $this->level += $level;
        } else {
            echo $level;
        }
    }

    public function addLimit(int $limit): void
    {
        if ($limit > 3) {
            $this->limit += $limit;
        } else {
            echo $limit;
        }
    }

    public function total(): int
    {
        return $this->index + $this->level + $this->limit;
    }

    public function describe(): string
    {
        $parts = [];
        $parts[] = (string) $this->index;
        $parts[] = (string) $this->level;
        $parts[] = (string) $this->limit;
        return implode(",", $parts);
    }
}
