<?php
function total(array $values): int {
    $sum = 0;
    foreach ($values as $value) {
        $sum += $value;
    }
    return $sum;
}

<?php
namespace Demo;

function active(array $users): array {
    return array_values(array_filter($users, function ($user) {
        return $user["active"] === true;
    }));
}

<?php
function label(int $count): string {
    if ($count === 0) {
        return "empty";
    }
    return "count:" . $count;
}

echo label(2);

<?php
class Counter {
    public static function from(int $value): self {
        $counter = new self();
        $counter->value = $value;
        return $counter;
    }
}
